%% ============================================================
%% OpenDocAgent — Professional Document Style
%% Applied automatically when format: pdf (article/report)
%% ============================================================

%% --- Core typography packages ---
\usepackage{lmodern}
\usepackage[T1]{fontenc}
\usepackage{microtype}

%% --- Layout and geometry helpers ---
\usepackage{graphicx}
\usepackage{float}
\usepackage{array}
\usepackage{ragged2e}

%% --- Tables ---
\usepackage{booktabs}
\usepackage{colortbl}
\usepackage{tabularx}
\usepackage{longtable}

%% --- Colors ---
\usepackage{xcolor}

%% --- Brand Color Palette ---
\definecolor{odaNavy}{HTML}{1F3A5F}
\definecolor{odaBlue}{HTML}{2E75B6}
\definecolor{odaSkyBlue}{HTML}{BDD7EE}
\definecolor{odaGray}{HTML}{595959}
\definecolor{odaLightGray}{HTML}{F2F2F2}
\definecolor{odaOrange}{HTML}{ED7D31}
\definecolor{odaRule}{HTML}{2E75B6}

%% --- Section heading styles (titlesec) ---
\usepackage{titlesec}

\titleformat{\section}
  {\large\bfseries\color{odaNavy}}
  {\thesection}{1em}{}
  [\color{odaRule}\vspace{2pt}\hrule height 0.8pt\vspace{2pt}]

\titleformat{\subsection}
  {\normalsize\bfseries\color{odaBlue}}
  {\thesubsection}{1em}{}

\titleformat{\subsubsection}
  {\normalsize\itshape\color{odaGray}}
  {\thesubsubsection}{1em}{}

\titlespacing*{\section}{0pt}{1.5ex plus 0.5ex minus 0.2ex}{0.8ex}
\titlespacing*{\subsection}{0pt}{1.2ex plus 0.3ex minus 0.2ex}{0.5ex}

%% --- Header and Footer (fancyhdr) ---
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}

%% Header: left = section, right = document title
\fancyhead[L]{\small\color{odaGray}\nouppercase{\leftmark}}
\fancyhead[R]{\small\bfseries\color{odaNavy}OpenDocAgent}

%% Header rule in brand color
\renewcommand{\headrulewidth}{0.6pt}
\renewcommand{\headrule}{%
  \hbox to\headwidth{\color{odaNavy}\leaders\hrule height \headrulewidth\hfill}}

%% Footer: center = page number
\fancyfoot[C]{\small\color{odaGray}\thepage}
\fancyfoot[R]{\small\color{odaLightGray!70!odaGray}\textit{Confidential}}
\renewcommand{\footrulewidth}{0pt}

%% Plain pages (TOC, chapter starts) also get footer
\fancypagestyle{plain}{%
  \fancyhf{}
  \fancyfoot[C]{\small\color{odaGray}\thepage}
  \renewcommand{\headrulewidth}{0pt}
}

%% --- Table styling ---
\setlength{\tabcolsep}{8pt}
\renewcommand{\arraystretch}{1.4}

%% --- List spacing ---
\usepackage{enumitem}
\setlist[itemize]{itemsep=2pt,topsep=4pt}
\setlist[enumerate]{itemsep=2pt,topsep=4pt}
\setlist[description]{itemsep=2pt,topsep=4pt}
