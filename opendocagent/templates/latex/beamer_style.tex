%% --- Block colors ---

%% --- Remove navigation symbols (always) ---
\setbeamertemplate{navigation symbols}{}

%% --- Bullet point style ---
\setbeamertemplate{itemize item}{\color{odaBlue}\textbullet}
\setbeamertemplate{itemize subitem}{\color{odaBlue}{\small$\circ$}}
\setbeamertemplate{itemize subsubitem}{\color{odaGray}{\small$-$}}

%% --- Enumerate item style ---
\setbeamertemplate{enumerate item}{\color{odaNavy}\textbf{\insertenumlabel.}}
\setbeamertemplate{enumerate subitem}{\color{odaBlue}\insertenumlabel.}

%% --- Custom footline: author | title | page/total (suppressed on slide 1) ---
\setbeamertemplate{footline}{%
  \ifnum\insertframenumber=1
    % Suppressed on first slide
  \else
    \leavevmode%
    \hbox{%
      \begin{beamercolorbox}[wd=0.50\paperwidth,ht=2.5ex,dp=1ex,leftskip=0.5em,rightskip=0.5em]{palette secondary}%
        \usebeamerfont{footline}\insertshortauthor%
        \ifx\insertshortinstitute\empty\else\ \textbar\ \insertshortinstitute\fi%
      \end{beamercolorbox}%
      \begin{beamercolorbox}[wd=0.30\paperwidth,ht=2.5ex,dp=1ex,center]{palette secondary}%
        \usebeamerfont{footline}\insertshorttitle%
      \end{beamercolorbox}%
      \begin{beamercolorbox}[wd=0.20\paperwidth,ht=2.5ex,dp=1ex,rightskip=0.5em,right]{palette primary}%
        \usebeamerfont{footline}%
        \insertframenumber\,/\,\inserttotalframenumber%
      \end{beamercolorbox}%
    }%
    \vskip0pt%
  \fi
}
